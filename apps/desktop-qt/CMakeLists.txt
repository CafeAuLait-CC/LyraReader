cmake_minimum_required(VERSION 3.22)
project(LyraReaderDesktop VERSION 0.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Concurrent ShaderTools Widgets PrintSupport)
find_package(Threads REQUIRED)
find_package(Qt6Keychain 0.14 REQUIRED)
get_filename_component(LYRA_READER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LYRA_READER_RUST_PROFILE "debug" CACHE STRING "Rust build profile (debug or release)")
if(WIN32)
    set(RUST_LIBRARY_NAME lyra_reader_desktop_ffi.lib)
else()
    set(RUST_LIBRARY_NAME liblyra_reader_desktop_ffi.a)
endif()
set(LYRA_READER_RUST_LIBRARY "${LYRA_READER_ROOT}/target/${LYRA_READER_RUST_PROFILE}/${RUST_LIBRARY_NAME}" CACHE FILEPATH "Built Rust desktop static library")
add_library(lyra_reader_rust STATIC IMPORTED GLOBAL)
set_target_properties(lyra_reader_rust PROPERTIES IMPORTED_LOCATION "${LYRA_READER_RUST_LIBRARY}")
find_program(LYRA_CARGO_EXECUTABLE cargo REQUIRED)
set(LYRA_CARGO_PROFILE_ARGS)
if(LYRA_READER_RUST_PROFILE STREQUAL "release")
    list(APPEND LYRA_CARGO_PROFILE_ARGS --release)
elseif(NOT LYRA_READER_RUST_PROFILE STREQUAL "debug")
    message(FATAL_ERROR "LYRA_READER_RUST_PROFILE must be debug or release")
endif()
add_custom_target(lyra_reader_build
    COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${LYRA_READER_ROOT}/target"
        ${LYRA_CARGO_EXECUTABLE} build --locked -p lyra-reader-desktop-ffi ${LYRA_CARGO_PROFILE_ARGS}
    WORKING_DIRECTORY "${LYRA_READER_ROOT}"
    BYPRODUCTS "${LYRA_READER_RUST_LIBRARY}"
    COMMENT "Checking/building the Rust document service"
    VERBATIM)
add_dependencies(lyra_reader_rust lyra_reader_build)
add_library(lyrareader_shell STATIC src/ReaderController.cpp src/ReaderController.h src/PageItem.cpp src/PageItem.h src/WorkspaceController.cpp src/WorkspaceController.h src/CredentialStore.cpp src/CredentialStore.h src/AnnotationOverlay.cpp src/AnnotationOverlay.h src/PaperAppearance.cpp src/PaperAppearance.h src/PdfPrintJob.cpp src/PdfPrintJob.h src/RasterCache.cpp src/RasterCache.h src/NativeRaster.cpp src/NativeRaster.h src/PreviewController.cpp src/PreviewController.h)
target_include_directories(lyrareader_shell PUBLIC src)
target_link_libraries(lyrareader_shell PUBLIC Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Widgets Qt6::PrintSupport Qt6Keychain::Qt6Keychain lyra_reader_rust Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(lyrareader_shell PUBLIC userenv ws2_32 bcrypt ntdll)
elseif(UNIX AND NOT APPLE)
    target_link_libraries(lyrareader_shell PUBLIC m rt util)
endif()
qt_add_executable(LyraReader src/main.cpp)
target_link_libraries(LyraReader PRIVATE lyrareader_shell)
set(LYRA_READER_QML_FILES qml/Main.qml qml/LyraReaderTheme.qml qml/LyraToolButton.qml
    qml/ReaderChrome.qml qml/ReaderSidebar.qml qml/DocumentCanvas.qml qml/PageInteraction.qml
    qml/PageOverview.qml qml/PreviewPanel.qml qml/AnnotationsLayer.qml qml/AnnotationHandles.qml qml/AnnotationTools.qml qml/SearchPanel.qml qml/AiPanel.qml qml/ReaderDialogs.qml qml/ReaderLogic.js)
file(GLOB LYRA_READER_ICONS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/icons/*.svg")
foreach(ICON_PATH IN LISTS LYRA_READER_ICONS)
    file(RELATIVE_PATH ICON_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${ICON_PATH}")
    list(APPEND LYRA_READER_QML_FILES "${ICON_RELATIVE}")
endforeach()
set(LYRA_READER_APPEARANCE_FONT "${LYRA_READER_ROOT}/apps/macos/Resources/Fonts/LyraReaderSansSC-Regular.ttf")
set_source_files_properties("${LYRA_READER_APPEARANCE_FONT}" PROPERTIES QT_RESOURCE_ALIAS "fonts/LyraReaderSansSC-Regular.ttf")
list(APPEND LYRA_READER_QML_FILES "${LYRA_READER_APPEARANCE_FONT}")
qt_add_resources(LyraReader qml_resources PREFIX "/" FILES ${LYRA_READER_QML_FILES})
qt_add_shaders(LyraReader annotation_shaders BATCHABLE PREFIX "/shaders" BASE "shaders" FILES shaders/annotation.vert shaders/annotation.frag shaders/paper.vert shaders/paper.frag)
get_target_property(QT_CORE_LIBRARY_TYPE Qt6::Core TYPE)
if(QT_CORE_LIBRARY_TYPE STREQUAL "SHARED_LIBRARY")
    # System/shared Qt loads QML plugins dynamically; importing nonexistent static
    # plugin targets produces misleading warnings on Debian's split Qt packages.
    set_target_properties(LyraReader PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
endif()
install(TARGETS LyraReader RUNTIME DESTINATION bin)
install(FILES "${LYRA_READER_ROOT}/apps/macos/Resources/Fonts/OFL.txt" DESTINATION share/licenses/LyraReader/fonts)
install(DIRECTORY licenses/ DESTINATION share/licenses/LyraReader/qt)
if(UNIX AND NOT APPLE)
    install(FILES platform/linux/com.alex.lyrareader.desktop DESTINATION share/applications)
    install(FILES platform/linux/com.alex.lyrareader.svg DESTINATION share/icons/hicolor/scalable/apps)
    install(FILES "${LYRA_READER_ROOT}/vendor/pdfium-linux/lib/libpdfium.so" DESTINATION bin/lib)
    install(FILES "${LYRA_READER_ROOT}/vendor/pdfium-linux/LICENSE" DESTINATION share/licenses/LyraReader/pdfium)
    install(DIRECTORY "${LYRA_READER_ROOT}/vendor/pdfium-linux/licenses/" DESTINATION share/licenses/LyraReader/pdfium/dependencies)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_custom_target(LyraReaderCheckFixtures
        COMMAND ${Python3_EXECUTABLE} "${LYRA_READER_ROOT}/scripts/make-fixtures.py"
            "${CMAKE_CURRENT_BINARY_DIR}/fixtures" --long-pages 9 --long-name qt-nine-pages.pdf
        BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/fixtures/qt-nine-pages.pdf"
        COMMENT "Generating synthetic Qt reading fixtures"
        VERBATIM)
    qt_add_executable(LyraReaderChecks tests/ReaderChecks.cpp)
    add_dependencies(LyraReaderChecks LyraReaderCheckFixtures)
    target_link_libraries(LyraReaderChecks PRIVATE lyrareader_shell Qt6::Test)
    qt_add_resources(LyraReaderChecks check_qml_resources PREFIX "/" FILES ${LYRA_READER_QML_FILES} tests/MultiplyCheck.qml)
    qt_add_shaders(LyraReaderChecks check_annotation_shaders BATCHABLE PREFIX "/shaders" BASE "shaders" FILES shaders/annotation.vert shaders/annotation.frag shaders/paper.vert shaders/paper.frag)
    if(QT_CORE_LIBRARY_TYPE STREQUAL "SHARED_LIBRARY")
        set_target_properties(LyraReaderChecks PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
    endif()
    add_test(NAME LyraReaderChecks COMMAND LyraReaderChecks "${LYRA_READER_ROOT}")
    set_tests_properties(LyraReaderChecks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
